# function to transcribe the user's audio
def transcribe_audio(file):
    global chat_now,chat_prev, is_Speaking, re_event
    
    try:
        audio_file= open(file, "rb")
        # Translating the audio to English
        # transcript = openai.Audio.translate("whisper-1", audio_file)
        # Transcribe the audio to detected language
        transcript = AIeri_prop.speech_to_text.convert(
            file=audio_file,
            model_id="scribe_v1",
            language_code="th"
        )
        chat_ = transcript.text
        audio_file.close()
        
        if chat_ and len(chat_.strip()) > 0:
            chat_now = owner_name + " said " + chat_
            conversation.append({'role': 'user', 'content': chat_now})
            chat_prev.append(chat_now)
            print(f"Transcribed: {chat_}")
            
            # เรียก AI เพื่อตอบ
            threading.Thread(target=openai_answer).start()
        else:
            print("No speech detected")
            # Reset สถานะถ้าไม่มีเสียง
            is_Speaking = False
            re_event = 0
            
    except Exception as e:
        print(f"Error in transcribe_audio: {e}")
        # Reset สถานะเมื่อเกิดข้อผิดพลาด
        is_Speaking = False
        re_event = 0

def translate_text(text_p):
    global is_Speaking , active_eri , mp3time , time_ 
    import os
    import time as time_module
    
    # ใช้ lock เพื่อป้องกันการใช้ไฟล์เสียงพร้อมกัน
    with audio_lock:
        # สร้างชื่อไฟล์ที่ไม่ซ้ำกัน
        timestamp = str(int(time_module.time() * 1000))
        output_file = f"output_{timestamp}.mp3"
        
        try:
            # หยุดเพลงที่เล่นอยู่และปลดล็อคไฟล์
            if pygame.mixer.music.get_busy():
                pygame.mixer.music.stop()
            
            # ปลดล็อคไฟล์เสียงปัจจุบัน
            try:
                pygame.mixer.music.unload()
            except:
                pass
            
            # รอให้ระบบปลดล็อคไฟล์
            time_module.sleep(0.5)
            
            # สร้างไฟล์เสียงใหม่
            eri = gTTS(text=text_p, lang='th', slow=False)
            eri.save(output_file)
            
            # เล่นไฟล์เสียงด้วยชื่อใหม่
            pygame.mixer.music.load(output_file)
            mp3time = mp3.MP3(output_file).info.length
            pygame.mixer.music.play()
            
            active_eri = 2
            
            # ลบไฟล์เก่าหลังจากเล่นแล้ว (ถ้าได้)
            def cleanup_old_files():
                time_module.sleep(1)  # รอให้ไฟล์ใหม่เริ่มเล่น
                try:
                    # ลบไฟล์เก่าที่ไม่ใช้แล้ว
                    for file in os.listdir("."):
                        if file.startswith("output_") and file.endswith(".mp3") and file != output_file:
                            try:
                                os.remove(file)
                            except:
                                pass
                    # ลบไฟล์ output.mp3 เก่าด้วย
                    if os.path.exists("output.mp3"):
                        try:
                            os.remove("output.mp3")
                        except:
                            pass
                except:
                    pass
            
            # เรียกทำความสะอาดในเธรดแยก
            threading.Thread(target=cleanup_old_files, daemon=True).start()
            
        except Exception as e:
            print(f"Error in translate_text: {e}")
            active_eri = 0
    
    
    
    
        
    

    # Clear the text files after the assistant has finished speaking
   
    with open ("output.txt", "w") as f:
        f.truncate(0)
    with open ("chat.txt", "w") as f:
        f.truncate(0)
